\section[Motivation]{Motivation}
% =========================================================================

\begin{frame}{What is limited-angle reconstruction?}
  \begin{itemize}
    \item Tomography recovers a cross-section from X-ray projections taken
          at many source angles around the object.
    \item \textbf{Sufficient sampling.} For a fan-beam scan, a source arc
          of $180^{\circ}$ plus the fan angle already measures every line
          integral; a full $360^{\circ}$ rotation only re-measures each
          ray from the opposite side.
    \item \textbf{Limited-angle reconstruction (LAR)} uses a source arc
          \emph{below} that threshold. The measurements no longer
          determine the image: a wedge of spatial frequencies is never
          observed.
    \item Digital breast tomosynthesis (DBT) operates deep in this regime,
          with a source arc of only $[15^{\circ}, 50^{\circ}]$, so its
          reconstruction is severely ill-posed.
  \end{itemize}
\end{frame}

\begin{frame}{Why limited-angle?}
  \begin{columns}[T,onlytextwidth]
    \column{.55\textwidth}
    \begin{itemize}
      \item \textbf{Dose.} Fewer views means less ionizing radiation
            per scan.
      \item \textbf{Geometry.} In DBT the detector stays under the
            compressed breast while the X-ray tube sweeps a short arc; a
            full gantry rotation is mechanically impossible.
      \item \textbf{Acquisition time.} A short sweep is fast, limiting
            motion blur and patient discomfort.
      \item \textbf{Use cases.} DBT for early breast-cancer detection;
            also lung-nodule and kidney-stone imaging.
    \end{itemize}
    \column{.45\textwidth}
    \centering
    \begin{tikzpicture}[scale=0.9, every node/.style={font=\scriptsize}]
      % Detector (gray bar at the bottom)
      \draw[fill=gray!22, thick] (-2.3, -1.8) rectangle (2.3, -1.65);

\begin{frame}{Two reconstruction settings: 2D and 3D}
  The two-channel method is evaluated in both geometries:
  \begin{itemize}
    \item \textbf{2D fan-beam.} A single slice; cheap enough to sweep
          image resolution and noise level.
    \item \textbf{3D cone-beam.} The actual DBT geometry; full-volume
          reconstruction on anatomical phantoms.
  \end{itemize}
  Both are severely under-determined: the number of unknowns far exceeds
  the number of measurements, increasingly so at finer resolution.
  \begin{center}
    \small
    \begin{tabular}{lcc}
      \toprule
      Slice resolution & Unknowns (pixels) & Measurements (rays) \\
      \midrule
      $512^2$ & $262{,}144$ & $25{,}600$ \\
      $256^2$ & $65{,}536$  & $25{,}600$ \\
      $128^2$ & $16{,}384$  & $25{,}600$ \\
      \bottomrule
    \end{tabular}
  \end{center}
  \footnotesize
  Measurements $=$ $25$ views $\times$ $1024$ detector bins.
\end{frame}

\section[Setup]{The reconstruction setup}
% =========================================================================

\begin{frame}{Reconstruction with PDHG (Chambolle--Pock)}
  Minimize directional-TV and $\ell_1$ sparsity, subject to the
  projections matching the data within a tolerance, with $f \ge 0$.
  \begin{itemize}
    \item The objective is non-differentiable ($\ell_1$, TV) and
          hard-constrained: gradient descent does not apply.
    \item PDHG alternates a \emph{dual} step (pull the data residual
          toward the tolerance ball) and a \emph{primal} step (pull the
          image toward sparsity and $f \ge 0$). Cheap projections and
          proximal maps, no derivatives.
  \end{itemize}
  Step sizes $\tau$ and $\sigma$ are coupled through the operator norm
  $\|K\|$; see appendix.
\end{frame}

\begin{frame}{Single-channel data fidelity}
  Prior work\footnote{E.\,Y.\,Sidky et al., \emph{J.\ Med.\ Imaging}
  12(S1):S13013, 2025.} solves a constrained DTV\,/\,$\ell_1$ problem with
  a square-root Hanning filter on the sinogram residual:
  \[
    \min_{f \geq 0}\; \alpha_x \|\partial_x f\|_1 + \alpha_y \|\partial_y f\|_1
                    + \alpha_z \|\partial_z f\|_1 + \beta \|f\|_1
  \]
  \[
    \text{s.t.}\quad \|R_{[c]}(g - X f)\|_2 \leq \varepsilon\sqrt{|g|}.
  \]
  \begin{itemize}
    \item $R_{[c]} = \Hann(c)$ attenuates low spatial frequencies in the
          residual (noise suppression by design).
    \item Solved with PDHG and He--Yuan over-relaxation
          ($\rho \approx 1.75$).
  \end{itemize}
\end{frame}

\section[Bottleneck]{The problem}
% =========================================================================

\begin{frame}{What happens during a single-channel run}
  \centering
  \deckfig[0.60]{single_only_ladder_256_paper.png}
  \vspace{0.3em}
  \begin{itemize}\setlength\itemsep{0pt}\small
    \item Fine structure (calcifications, glandular speckling) settles fast.
    \item Large soft-tissue regions stay blurred, with a slowly-varying
          intensity error, through iter ${\sim}100\text{--}300$.
    \item This is the low-frequency convergence bottleneck.
  \end{itemize}
  \geomnote{2D fan-beam $\cdot$ 25 views $\cdot$ 50$^\circ$ arc $\cdot$ $256^2$ grid}
\end{frame}

\begin{frame}{Why does this happen?}
  In the detector-frequency domain, the data-fidelity weight is
  \[
    W(g - Xf), \qquad W = R_{[c]} = \Hann(c).
  \]
  \begin{itemize}
    \item PDHG corrects mode $k$ at a rate $\propto s_k^2$
          (squared singular values of $WX$).
    \item In LAR DBT, $s_k$ is already tiny for low $k$ (ill-conditioning).
    \item $\Hann$ shrinks low-$k$ further \emph{by design}.
    \item Stability bound is dictated by the \emph{largest} $s_k$
          (high frequencies).
  \end{itemize}
  \begin{alertblock}{Consequence}
    The optimizer is forced to use step sizes that under-drive the very
    modes that are hardest to recover.
  \end{alertblock}
\end{frame}

% =========================================================================
\section[Two-channel]{The idea: two-channel fidelity}
% =========================================================================

\begin{frame}{Split the sinogram into two bands}
  \[
    \min_{f \geq 0}\; \alpha_\bullet \|\partial_\bullet f\|_1 + \beta \|f\|_1
  \]
  \[
    \text{s.t.}\quad
    \begin{cases}
      \|\Rhi (g - Xf)\|_2 \le \varepsilon_{\mathrm{hi}} \sqrt{|g|}, \\[2pt]
      \|\Rlo (g - Xf)\|_2 \le \varepsilon_{\mathrm{lo}} \sqrt{|g|}.
    \end{cases}
  \]
  \begin{itemize}\small
    \item $\Rhi = \Hann(c)$ -- preserves the high-frequency channel from
          prior work.
    \item $\Rlo = \Hann(c_{\mathrm{lo}})$ with $c_{\mathrm{lo}} \gg c$ --
          a narrow-band low-pass channel that targets the slow modes.
    \item Independent dual variables $y_{\mathrm{hi}}, y_{\mathrm{lo}}$
          with independent step sizes
          $\sigma_{\mathrm{hi}}, \sigma_{\mathrm{lo}}$.
  \end{itemize}
  Step sizes follow the same balance heuristic as the single-channel
  paper: $\tau\,\sigma_{\mathrm{hi}}\,\|K\|^{2} \approx 1$, with
  $\sigma_{\mathrm{lo}} = 4 \sigma_{\mathrm{hi}}$ to drive the slow modes
  harder.
\end{frame}

\begin{frame}{The two-channel iteration}
  \centering
  \resizebox{\textwidth}{!}{%
  \begin{tikzpicture}[
    every node/.style={font=\footnotesize},
    box/.style={draw, rounded corners, align=center, fill=MainAccent!12,
                minimum height=1.0cm},
    hi/.style={draw, rounded corners, align=center, fill=MainAccent!22,
               minimum height=1.15cm, minimum width=4.1cm},
    lo/.style={draw, rounded corners, align=center, fill=SubAccent!22,
               minimum height=1.15cm, minimum width=4.1cm},
    arr/.style={-Latex, semithick},
  ]
    \node[box, minimum width=1.9cm] (img)  at (0,0)     {image\\$f$};
    \node[box, minimum width=2.8cm] (res)  at (3.2,0)   {residual\\$g - X f$};
    \node[hi]                       (hi)   at (8.0,1.35)
      {high band: $\Rhi$\\dual $y_{\mathrm{hi}}$,\;\; step $\sigma_{\mathrm{hi}}$};
    \node[lo]                       (lo)   at (8.0,-1.35)
      {low band: $\Rlo$\\dual $y_{\mathrm{lo}}$,\;\; step $\sigma_{\mathrm{lo}} = 4\,\sigma_{\mathrm{hi}}$};
    \node[box, minimum width=3.0cm] (prim) at (12.9,0)
      {primal update\\DTV $+\,\ell_1+ (f\!\ge\!0)$};

    \draw[arr] (img) -- (res);
    \draw[arr] (res.east) -- ++(0.45,0) |- (hi.west);
    \draw[arr] (res.east) -- ++(0.45,0) |- (lo.west);
    \draw[arr] (hi.east) -| (prim.north);
    \draw[arr] (lo.east) -| (prim.south);
    \draw[arr] (prim.south) -- ++(0,-1.7) -| (img.south)
      node[pos=0.5, fill=white, inner sep=2pt, font=\scriptsize]
        {repeat each iteration};
  \end{tikzpicture}}
  \vspace{0.3em}
  \begin{itemize}\setlength\itemsep{2pt}\small
    \item The data residual is split into a high- and a low-frequency band.
    \item Each band carries its own dual variable and its own step size.
    \item The low band is driven $4\times$ harder
          ($\sigma_{\mathrm{lo}} = 4\,\sigma_{\mathrm{hi}}$), so the slow
          modes receive the larger correction.
    \item The primal update and the image itself are unchanged from
          single-channel.
  \end{itemize}
\end{frame}

\section[Conclusion]{Conclusion}
% =========================================================================

\begin{frame}{Summary}
  \begin{itemize}
    \item LAR reconstruction has a \emph{mode-dependent} convergence
          bottleneck: low spatial frequencies are starved of corrective
          drive because the PDHG step size is bound by high-frequency
          stability.
    \item Splitting the data fidelity into a low-pass and a high-pass
          channel, each with its own dual variable and step size,
          accelerates low-frequency settling without breaking stability.
    \item Confirmed at a clinical $50^\circ$ arc on the 2D breast phantom
          (iter-$100$ image-RMSE $26\%$ below single-channel) and the 3D
          analytic breast in DBT geometry ($9\%$ at iter $200$). The gain
          persists at wider arcs on the head and jaw phantoms.
  \end{itemize}
\end{frame}

\begin{frame}{Future work}
  \begin{itemize}
    \item \textbf{Why just two channels?} Extend the two-band split to a
          multi-channel band-pass decomposition. More channels could
          give finer per-mode control, at the cost of additional dual
          variables and a more careful step-size schedule. Manuscript
          in preparation.
    \item \textbf{Convergence guarantees.} Derive the per-band step
          sizes from a weighted power iteration so the empirical
          balance heuristic connects to the product-space PDHG
          convergence guarantee, both for the two-channel case and any
          multi-channel extension.
    \item \textbf{Tuning the low-frequency constraint.} The per-band
          data-tolerance ratio that minimises image RMSE is not the
          same on 2D and 3D phantoms. Understanding what problem
          features control it is open.
  \end{itemize}
\end{frame}

\section[Backup]{Backup: per-band tolerance tuning}
% =========================================================================

\begin{frame}{The per-band tolerance ratio}
  Each channel constrains its band of the data residual to lie inside an
  $\ell_2$ ball,
  \[
    \|\Rhi(g - Xf)\|_2 \le \varepsilon_{\mathrm{hi}}\sqrt{|g|},
    \qquad
    \|\Rlo(g - Xf)\|_2 \le \varepsilon_{\mathrm{lo}}\sqrt{|g|}.
  \]
  The ratio $r = \varepsilon_{\mathrm{lo}} / \varepsilon_{\mathrm{hi}}$
  sets how tightly the low-frequency residual is held relative to the
  high-frequency residual:
  \begin{itemize}\setlength\itemsep{0pt}\small
    \item Small $r$ tightens the LF data fit; the recon is pulled
          closer to the data in the low-frequency band.
    \item Large $r$ loosens the LF constraint; the LF prior (DTV,
          $\ell_1$, non-negativity) carries more weight.
    \item Changing $r$ moves the \emph{fixed point} of the optimization,
          not just the trajectory. The original paper used $r = 1.25$;
          the next slide sweeps it to see how much performance is
          available.
  \end{itemize}
\end{frame}

\begin{frame}{Tolerance-ratio sweep across phantoms}
  \centering
  \deckfig[0.55]{design_space_iter100.png}
  \vspace{0.2em}
  \begin{itemize}\setlength\itemsep{0pt}\small
    \item 2D breast: clear interior optimum near $r \approx 15$
          ($-45\%$ vs single-channel); paper $r{=}1.25$ leaves $\sim 20\%$.
    \item 3D analytic breast: flat plateau at $-34\%$; paper config is
          already near the optimum.
    \item The optimum does not transfer between dimensions. Band-cutoff
          is empirically robust on both (sweep omitted).
  \end{itemize}
  \geomnote{Iter-$100$ image-RMSE reduction vs $r$ $\cdot$ 25 views $\cdot$ 50$^\circ$ arc}
\end{frame}
